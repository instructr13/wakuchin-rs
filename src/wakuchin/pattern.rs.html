13/// A compiled pattern for detecting hits.
14///
15/// Shared across workers: it is `Sync` and holds no per-search scratch space.
16/// Each worker makes its own [`Matcher`], which owns that scratch space.
17///
18/// # Why not `regex::Regex`
19///
20/// `regex`'s `is_match` keeps its scratch space in an internal pool and checks
21/// it out on every call. The pool has a fast path for the thread that owns it,
22/// and a mutex for every other thread.
23///
24/// Taking a pattern *string* rather than an already-built regex is deliberate.
25/// Rebuilding a `regex::Regex` from its `as_str()` would silently discard any
26/// `RegexBuilder` configuration the caller had applied, with no error. Every
27/// setting that changes what a pattern *means* can be written inline in the
28/// pattern instead -- `(?i)`, `(?s)`, `(?m)`, `(?x)`, `(?-u)`.
29#[derive(Clone)]
30pub struct Pattern {
31  regex: Regex,
32
33  /// `meta::Regex` does not keep the pattern it was built from.
34  source: Box<str>,
35}
36
37impl Pattern {
38  /// Compiles `pattern`.
39  ///
40  /// Matching semantics follow `regex::bytes::Regex`: leftmost-first, empty
41  /// matches allowed to land between UTF-8 code units, and the same size
42  /// limits.
43  ///
44  /// # Errors
45  ///
46  /// [`WakuchinError::Pattern`] if `pattern` is not a valid regular
47  /// expression, or needs more memory than the limits allow.
48  pub fn new(pattern: &str) -> Result<Self, WakuchinError> {
49    // Mirrors what `regex::bytes::RegexBuilder` configures, so a pattern
50    // behaves the same here as it would through the `regex` crate.
51    let config = meta::Config::new()
52      .nfa_size_limit(Some(10 * (1 << 20)))
53      .hybrid_cache_capacity(2 * (1 << 20))
54      .match_kind(MatchKind::LeftmostFirst)
55      .utf8_empty(false);
56
57    Regex::builder()
58      .configure(config)
59      .syntax(syntax::Config::new().utf8(false))
60      .build(pattern)
61      .map(|regex| Self {
62        regex,
63        source: pattern.into(),
64      })
65      .map_err(|err| {
66        // `BuildError`'s own `Display` is just "error parsing pattern 0"; the
67        // part a user can act on is the syntax error underneath it.
68        WakuchinError::Pattern(
69          err
70            .syntax_error()
71            .map_or_else(|| err.to_string(), ToString::to_string),
72        )
73      })
74  }
75
76  /// The pattern this was compiled from.
77  #[must_use]
78  pub fn as_str(&self) -> &str {
79    &self.source
80  }
81
82  /// A matcher with its own scratch space. Make one per thread.
83  #[must_use]
84  pub fn matcher(&self) -> Matcher<'_> {
85    Matcher {
86      pattern: self,
87      cache: self.regex.create_cache(),
88    }
89  }
90
91  /// Whether `candidate` matches.
92  ///
93  /// Allocates scratch space on every call. Fine for a one-off check; for a
94  /// loop, keep a [`Matcher`].
95  #[must_use]
96  pub fn is_match(&self, candidate: &[u8]) -> bool {
97    self.matcher().is_match(candidate)
98  }
99}

139/// A [`Pattern`] plus the scratch space one search needs.
140///
141/// Make one per worker and keep it for the whole loop. `is_match` takes
142/// `&mut self`, so two threads cannot use one matcher at the same time; that
143/// is the point -- the alternative is what `regex` does, which is to reacquire
144/// the scratch space on every single call.
145#[derive(Clone, Debug)]
146pub struct Matcher<'pattern> {
147  pattern: &'pattern Pattern,
148  cache: meta::Cache,
149}
150
151impl Matcher<'_> {
152  /// Whether `candidate` matches.
153  #[inline(always)]
154  pub fn is_match(&mut self, candidate: &[u8]) -> bool {
155    // `earliest(true)` stops at the first match instead of finding the
156    // leftmost-first one, which is all a boolean answer needs.
157    let input = Input::new(candidate).earliest(true);
158
159    self
160      .pattern
161      .regex
162      .search_half_with(&mut self.cache, &input)
163      .is_some()
164  }
165
166  #[must_use]
167  pub fn pattern(&self) -> &Pattern {
168    self.pattern
169  }
170}

180  #[test]
181  fn matches_like_the_regex_crate() -> Result<(), Box<dyn Error>> {
182    let cases: &[(&str, &[(&str, bool)])] = &[
183      (
184        "^[WKCN]+$",
185        &[("WKCN", true), ("WKCNX", false), ("わくちん", false)],
186      ),
187      ("^WKNCWKNC$", &[("WKNCWKNC", true), ("WKNCWKNCW", false)]),
188      ("WKCN", &[("NNWKCNCC", true), ("NNWKNCCC", false)]),
189      ("(?i)wkcn", &[("WKCN", true)]),
190      (
191        "(WKNCWKNC|WCKNWCKN)",
192        &[("WCKNWCKN", true), ("WKCNWKCN", false)],
193      ),
194    ];
195
196    for (pattern, expectations) in cases {
197      let compiled = Pattern::new(pattern)?;
198      let mut matcher = compiled.matcher();
199
200      assert_eq!(compiled.as_str(), *pattern);
201
202      for (candidate, expected) in *expectations {
203        assert_eq!(
204          matcher.is_match(candidate.as_bytes()),
205          *expected,
206          "/{pattern}/ against {candidate:?}"
207        );
208        assert_eq!(compiled.is_match(candidate.as_bytes()), *expected);
209      }
210    }
211
212    Ok(())
213  }

215  #[test]
216  fn a_pattern_is_shareable_and_a_matcher_is_owned() {
217    const fn assert_send_sync<T: Send + Sync>() {}
218
219    // The pattern is shared by every worker; each moves its own matcher into
220    // its thread. Exclusive use of a matcher is enforced by `is_match` taking
221    // `&mut self`, not by the matcher being `!Sync` -- the cache it wraps
222    // happens to be `Sync`.
223    assert_send_sync::<Pattern>();
224    assert_send_sync::<Matcher<'_>>();
225  }

237  /// A matcher reused across candidates has to keep answering correctly -- the
238  /// cache it carries is exactly the state `regex` would have thrown away
239  /// after every call.
240  #[test]
241  fn a_reused_matcher_stays_correct() -> Result<(), Box<dyn Error>> {
242    let pattern = Pattern::new("^WK")?;
243    let mut matcher = pattern.matcher();
244    let mut generator = Generator::new(2);
245
246    for _ in 0..10_000 {
247      generator.shuffle();
248
249      assert_eq!(
250        matcher.is_match(generator.bytes()),
251        generator.as_str().starts_with("WK")
252      );
253    }
254
255    Ok(())
256  }
