26/// Largest `times` a research run accepts.
27///
28/// The candidate space has `(4n)!/(n!)^4` distinct arrangements. At `n = 8`
29/// that is 99,561,092,450,391,000, and at `n = 9` it stops fitting in a `u64`.
30pub const MAX_TIMES: usize = 8;

32/// Generates random wakuchin candidates into a buffer it owns.
33#[derive(Clone, Debug)]
34pub struct Generator {
35  buf: Vec<u8>,
36  rng: fastrand::Rng,
37}
38
39impl Generator {
40  /// # Arguments
41  ///
42  /// * `times` - wakuchin times n
43  ///
44  /// # Examples
45  ///
46  /// ```rust
47  /// use wakuchin::Generator;
48  ///
49  /// let mut generator = Generator::new(2);
50  ///
51  /// generator.shuffle();
52  ///
53  /// assert_eq!(generator.bytes().len(), 8);
54  /// ```
55  #[must_use]
56  pub fn new(times: usize) -> Self {
57    Self {
58      buf: WAKUCHIN_BYTES.repeat(times),
59      // Owned rather than the `fastrand` free functions, which reach through a
60      // thread local on every call.
61      rng: fastrand::Rng::new(),
62    }
63  }
64
65  /// Replaces the current candidate with a fresh one. Allocates nothing.
66  #[inline]
67  pub fn shuffle(&mut self) {
68    self.rng.shuffle(&mut self.buf);
69  }
70
71  /// The current candidate's bytes.
72  #[inline]
73  #[must_use]
74  pub fn bytes(&self) -> &[u8] {
75    &self.buf
76  }
77
78  /// The current candidate.
79  ///
80  /// # Panics
81  ///
82  /// Never in practice: the buffer only ever holds a permutation of
83  /// [`WAKUCHIN_BYTES`], which is ASCII.
84  #[inline]
85  #[must_use]
86  pub fn as_str(&self) -> &str {
87    str::from_utf8(&self.buf).expect("wakuchin candidates are ASCII")
88  }
89}

91/// Generate a randomized wakuchin string.
92///
93/// # Arguments
94///
95/// * `times` - wakuchin times n
96///
97/// # Returns
98///
99/// * `String` - randomized wakuchin string
100///
101/// # Examples
102///
103/// ```rust
104/// use wakuchin::gen;
105/// use wakuchin::symbol;
106///
107/// let wakuchin = gen(3);
108///
109/// assert_eq!(wakuchin.len(), 12);
110///
111/// let mut wakuchin_w_count: u32 = 0;
112/// let mut wakuchin_k_count: u32 = 0;
113/// let mut wakuchin_c_count: u32 = 0;
114/// let mut wakuchin_n_count: u32 = 0;
115///
116/// for c in wakuchin.chars() {
117///   match c {
118///     symbol::WAKUCHIN_W => wakuchin_w_count += 1,
119///     symbol::WAKUCHIN_K => wakuchin_k_count += 1,
120///     symbol::WAKUCHIN_C => wakuchin_c_count += 1,
121///     symbol::WAKUCHIN_N => wakuchin_n_count += 1,
122///     _ => panic!("Unexpected character: {}", c),
123///   }
124/// }
125///
126/// assert_eq!(wakuchin_w_count, 3);
127/// assert_eq!(wakuchin_k_count, 3);
128/// assert_eq!(wakuchin_c_count, 3);
129/// assert_eq!(wakuchin_n_count, 3);
130/// ```
131#[inline]
132pub fn gen(times: usize) -> Cow<'static, str> {
133  let mut wakuchin = WAKUCHIN_BYTES.repeat(times);
134
135  fastrand::shuffle(&mut wakuchin);
136
137  String::from_utf8(wakuchin)
138    .expect("wakuchin candidates are ASCII")
139    .into()
140}

142/// Generate a vector of randomized wakuchin string.
143/// This function is useful when you want to generate multiple wakuchin strings.
144///
145/// # Arguments
146///
147/// * `len` - length of vector to generate
148/// * `times` - wakuchin times n
149///
150/// # Returns
151///
152/// * `Vec<String>` - vector of randomized wakuchin string
153///
154/// # Examples
155///
156/// ```rust
157/// use wakuchin::gen_vec;
158/// use wakuchin::symbol;
159///
160/// let wakuchin_vec = gen_vec(3, 3);
161///
162/// assert_eq!(wakuchin_vec.len(), 3);
163///
164/// let mut wakuchin_w_count: u32 = 0;
165/// let mut wakuchin_k_count: u32 = 0;
166/// let mut wakuchin_c_count: u32 = 0;
167/// let mut wakuchin_n_count: u32 = 0;
168///
169/// for wakuchin in wakuchin_vec {
170///   assert_eq!(wakuchin.len(), 12);
171///
172///   for c in wakuchin.chars() {
173///     match c {
174///       symbol::WAKUCHIN_W => wakuchin_w_count += 1,
175///       symbol::WAKUCHIN_K => wakuchin_k_count += 1,
176///       symbol::WAKUCHIN_C => wakuchin_c_count += 1,
177///       symbol::WAKUCHIN_N => wakuchin_n_count += 1,
178///       _ => panic!("Unexpected character: {}", c),
179///     }
180///   }
181/// }
182///
183/// assert_eq!(wakuchin_w_count, 9);
184/// assert_eq!(wakuchin_k_count, 9);
185/// assert_eq!(wakuchin_c_count, 9);
186/// assert_eq!(wakuchin_n_count, 9);
187/// ```
188#[inline]
189pub fn gen_vec(len: usize, times: usize) -> Vec<Cow<'static, str>> {
190  (0..len).map(|_| gen(times)).collect()
191}

193/// Check if a string is a internally used wakuchin string.
194///
195/// # Arguments
196///
197/// * `wakuchin` - internal wakuchin string to check
198///
199/// # Returns
200///
201/// * `bool` - true if internal wakuchin string is valid
202///
203/// # Examples
204///
205/// ```rust
206/// use wakuchin::validate;
207///
208/// assert!(validate("WKCN"));
209/// assert!(!validate("わくちん"));
210/// assert!(!validate("WKCNX"));
211/// ```
212pub fn validate(wakuchin: &str) -> bool {
213  wakuchin.chars().all(|c| symbol::WAKUCHIN.contains(&c))
214}
215
216/// Check whether the string is a valid wakuchin characters.
217///
218/// # Arguments
219///
220/// * `wakuchin` - wakuchin string to check
221///
222/// # Returns
223///
224/// * `bool` - true if wakuchin string is valid
225///
226/// # Examples
227///
228/// ```rust
229/// use wakuchin::validate_external;
230///
231/// assert!(validate_external("わくちん"));
232/// assert!(!validate_external("WKCN"));
233/// assert!(!validate_external("わくうちん"));
234/// assert!(!validate_external("WKCNX"));
235/// ```
236pub fn validate_external(wakuchin: &str) -> bool {
237  wakuchin
238    .chars()
239    .all(|c| symbol::WAKUCHIN_EXTERNAL.contains(&c))
240}

242/// Check wakuchin candidate bytes with a matcher.
243///
244/// Takes the [`pattern::Matcher`] rather than the [`pattern::Pattern`] so the
245/// cost is visible in the signature: the matcher carries the regex engine's
246/// scratch space, and building that is far more expensive than a match. Make
247/// one per thread and keep it for the whole loop, which is what
248/// [`worker::run_par`] and [`worker::run_seq`] do.
249///
250/// # Arguments
251///
252/// * `chars` - wakuchin candidate bytes to check
253/// * `matcher` - matcher to check with
254///
255/// # Returns
256///
257/// * `bool` - true if the candidate matches
258///
259/// # Examples
260///
261/// ```rust
262/// use wakuchin::check;
263/// use wakuchin::pattern::Pattern;
264///
265/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
266/// let pattern = Pattern::new(r"^[WKCN]+$")?;
267/// let mut matcher = pattern.matcher();
268///
269/// assert!(check(b"WKCN", &mut matcher));
270/// assert!(!check("わくちん".as_bytes(), &mut matcher));
271/// assert!(!check(b"WKCNX", &mut matcher));
272/// #
273/// #   Ok(())
274/// # }
275/// ```
276#[inline(always)]
277pub fn check(chars: &[u8], matcher: &mut Matcher<'_>) -> bool {
278  matcher.is_match(chars)
279}

394  #[test]
395  fn test_generator_reuse_stays_uniform() {
396    /// Multiset arrangements of "WKCN" repeated twice: 8!/(2!)^4.
397    const ARRANGEMENTS: usize = 2520;
398
399    /// A hundred expected observations per bin keeps the chi-square
400    /// approximation comfortable.
401    const SAMPLES: usize = ARRANGEMENTS * 100;
402
403    /// Chi-square, 2519 degrees of freedom, upper tail 1e-6 (~2872).
404    const CRITICAL: f64 = 2900.0;
405
406    let mut generator = Generator::new(2);
407    let mut counts = std::collections::HashMap::new();
408
409    for _ in 0..SAMPLES {
410      generator.shuffle();
411
412      *counts
413        .entry(generator.as_str().to_owned())
414        .or_insert(0_usize) += 1;
415    }
416
417    assert_eq!(
418      counts.len(),
419      ARRANGEMENTS,
420      "not every arrangement was produced"
421    );
422
423    let expected = SAMPLES as f64 / ARRANGEMENTS as f64;
424    let chi_square: f64 = counts
425      .values()
426      .map(|&observed| {
427        let deviation = observed as f64 - expected;
428
429        deviation * deviation / expected
430      })
431      .sum();
432
433    assert!(
434      chi_square < CRITICAL,
435      "chi-square {chi_square} over {ARRANGEMENTS} arrangements suggests a non-uniform distribution (critical value {CRITICAL})"
436    );
437  }
